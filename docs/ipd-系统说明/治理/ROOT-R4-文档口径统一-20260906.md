## 4. 根治方案 3 步骤详述

### 步骤 1：CI 钩子 `docs-link-check.yml`（P0-9）

**落点**：`.github/workflows/docs-link-check.yml`（周日 02:00 跑 + PR 触发）

**校验维度**（4 维度三向校验）：

1. **字段名校验**：文档中出现的 Java 字段名（如 `actualReceipts` / `gateReviewerCount`）必须在 `ruoyi-modules/ruoyi-ipd/src/main/java` 中 `rg -l` 命中 ≥1 处
2. **枚举值校验**：文档中出现的枚举字面量（如 `SUBMITTED` / `TRIAGING` / `PLANNED`）必须在对应 enum 类中存在
3. **权限码校验**：文档中出现的权限码字面量（如 `ipd:gate-review:list`）必须在 `IpdPermissionCode` 常量类中登记
4. **AC 编号校验**：文档中出现的 `AC-INC-N` 编号必须在 `acceptance-matrix.json`（步骤 2）中有行

**Python 钩子骨架**（落点 `.claude/helpers/docs-link-validate.cjs`，与勘察卡 §7.1 第 5 行 `docs-link-validate.cjs` 对齐）：

```python
#!/usr/bin/env python3
# .claude/helpers/docs-link-validate.py
"""
文档-代码双向引用校验（ROOT-R4 P0-9）
四种校验：字段名 / 枚举值 / 权限码 / AC 编号
输出：lint-reports/docs-link-YYYY-MM-DD.md + GitHub Actions annotation
"""
import json, re, sys
from pathlib import Path
from typing import Iterable

REPO_ROOT = Path(__file__).resolve().parents[2]
IPD_JAVA = REPO_ROOT / "ruoyi-modules" / "ruoyi-ipd" / "src" / "main" / "java"
DOCS = REPO_ROOT / "docs" / "ipd-系统说明"
MATRIX = DOCS / "治理" / "acceptance-matrix.json"

def collect_doc_tokens() -> dict[str, set[str]]:
    """扫所有 docs/ipd-系统说明/**/*.md，提取疑似 Java 字段名 / 枚举值 / 权限码。"""
    fields: set[str] = set()
    enums:  set[str] = set()
    codes:  set[str] = set()
    acs:    set[str] = set()
    field_re = re.compile(r'\b([a-z][a-zA-Z0-9]{4,40})\b')      # camelCase 字段
    enum_re  = re.compile(r'`([A-Z][A-Z_]{2,30})`')              # UPPER_SNAKE 枚举
    code_re  = re.compile(r'`(ipd:[a-z-]+:[a-z-]+)`')           # 权限码格式
    ac_re    = re.compile(r'AC-(INC|EXT|MIN)-\d+')               # AC 编号
    for md in DOCS.rglob("*.md"):
        for line in md.read_text(encoding="utf-8").splitlines():
            fields.update(field_re.findall(line))
            enums.update(enum_re.findall(line))
            codes.update(code_re.findall(line))
            acs.update(ac_re.findall(line))
    return {"fields": fields, "enums": enums, "codes": codes, "acs": acs}

def load_code_facts() -> dict[str, set[str]]:
    """从 Java 源码 + IpdPermissionCode 收集真实存在的字段/枚举/权限码。"""
    java_fields: set[str] = set()
    enums:       set[str] = set()
    codes:       set[str] = set()
    if IPD_JAVA.exists():
        for java in IPD_JAVA.rglob("*.java"):
            text = java.read_text(encoding="utf-8", errors="ignore")
            java_fields.update(re.findall(r'private\s+\S+\s+([a-zA-Z][a-zA-Z0-9]+)\s*[=;]', text))
            for m in re.finditer(r'enum\s+\w+\s*\{([^}]+)\}', text, re.S):
                enums.update(re.findall(r'\b([A-Z][A-Z_]{2,30})\b', m.group(1)))
    code_const = IPD_JAVA / "common" / "IpdPermissionCode.java"
    if code_const.exists():
        codes.update(re.findall(r'"(ipd:[a-z-]+:[a-z-]+)"', code_const.read_text(encoding="utf-8", errors="ignore")))
    return {"fields": java_fields, "enums": enums, "codes": codes}

def load_matrix() -> set[str]:
    if not MATRIX.exists():
        return set()
    return {row["ac_id"] for row in json.loads(MATRIX.read_text(encoding="utf-8"))["rows"]}

def main() -> int:
    doc = collect_doc_tokens()
    code = load_code_facts()
    matrix = load_matrix()
    missing = {
        "fields": doc["fields"] - code["fields"],
        "enums":  doc["enums"]  - code["enums"],
        "codes":  doc["codes"]  - code["codes"],
        "acs":    doc["acs"]    - matrix,
    }
    total_missing = sum(len(v) for v in missing.values())
    report = DOCS / "lint-reports" / f"docs-link-{date.today().isoformat()}.md"
    report.parent.mkdir(parents=True, exist_ok=True)
    report.write_text(render_markdown(missing), encoding="utf-8")
    print(f"缺失引用 {total_missing} 处：field={len(missing['fields'])} enum={len(missing['enums'])} code={len(missing['codes'])} ac={len(missing['acs'])}")
    return 1 if total_missing > 0 else 0  # WARN（exit 0）vs ERROR（exit 1）由 CI 配置决定

if __name__ == "__main__":
    sys.exit(main())
```

**CI 行为**：

- **WARN 级**（默认）：缺失项写报告但不阻断 PR（避免误杀）
- **ERROR 级**（P0-9 落地 30 天后升级）：缺失项 ≥ 阈值阻断 PR
- **报告归档**：`docs/ipd-系统说明/lint-reports/YYYY-MM-DD.md` + 写入周治理卡

**与现有 hook 协作**：

- `sensitive-field-guard.cjs`（ERROR 阻断）→ 保护 secret/credential
- `pom-edit-hint.cjs`（WARN）→ 提示 pom 同步项
- `block-dangerous-git.sh`（ERROR 阻断）→ 保护 git 危险命令
- **新增 `docs-link-validate.py`（WARN→ERROR 渐进）**→ 保护文档-代码口径
